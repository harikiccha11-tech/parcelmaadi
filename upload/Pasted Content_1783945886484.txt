        calculateDiscount: function(subtotal, codes, first, referral, festival, corporate, bulk) {
            let total = 0, applied = [];

            if (first) { let d = this.discounts['FIRST_ORDER']; let amt = Math.min(subtotal * d.value / 100, d.max); total += amt; applied.push({ type: 'FIRST_ORDER', amount: amt }); }

            if (festival) { let d = this.discounts['FESTIVAL']; let amt = Math.min(subtotal * d.value / 100, d.max); total += amt; applied.push({ type: 'FESTIVAL', amount: amt }); }
            if (corporate) { let d = this.discounts['CORPORATE']; let amt = Math.min(subtotal * d.value / 100, d.max); total += amt; applied.push({ type: 'CORPORATE', amount: amt }); }
            if (bulk) { let d = this.discounts['BULK_ORDER']; let amt = Math.min(subtotal * d.value / 100, d.max); total += amt; applied.push({ type: 'BULK_ORDER', amount: amt }); }

            if (codes && codes.length) {
                for (let code of codes) {
                    let d = this.discounts[code];
                    if (d) {
                        let amt = d.type === 'percentage' ? Math.min(subtotal * d.value / 100, d.max) : Math.min(d.value, d.max);
                        total += amt;
                        applied.push({ type: code, amount: amt });
                    }
                }
            }

            const v = this.vehicles[vehicleType];
            if (!v) return { error: 'Invalid vehicle' };

            const base = this.getSlabRate(vehicleType, distanceKM);
            const wait = this.getWaitingCharge(vehicleType, waitMinutes || 0);
            let subtotal = base + wait;

            const timeSurge = this.getTimeSurge(timeHour);
            let afterTime = Math.round(subtotal * timeSurge.multiplier);

            const trafficMulti = this.trafficSurge[trafficStatus] || 1.0;
            let afterTraffic = Math.round(afterTime * trafficMulti);

            const disc = this.calculateDiscount(
                afterTraffic,
                discountCodes || [],
                isFirstOrder || false,
                isReferral || false,
                isFestival || false,
                isCorporate || false,
                isBulkOrder || false
            );

            let final = Math.max(0, Math.round(afterTraffic - disc.total));
            const rounded = this.roundUp(final);
            const commission = Math.round(rounded * v.commission / 100);
            const payout = rounded - commission;

            return {
                success: true,
                vehicle: { type: vehicleType, name: v.name, capacity: v.capacity, icon: v.icon },
                distance: distanceKM,
                time: { hour: timeHour, surge: timeSurge.multiplier, label: timeSurge.label },
                traffic: { status: trafficStatus, multiplier: trafficMulti },
                pricing: {
                    basePrice: base,
                    waitingCharge: wait,
                    subtotal: subtotal,
                    afterTimeSurge: afterTime,
                    afterTrafficSurge: afterTraffic,
                    discount: disc.total,
                    discountDetails: disc.applied,
                    finalPrice: final,
                    roundedPrice: rounded,
                    commission: commission,
                    driverPayout: payout,
                    porterPrice: Math.round(rounded * 1.025),

                    'You Save': '₹' + Math.round(rounded * 0.025),
                    'Commission': '₹' + commission,
                    'Driver Payout': '₹' + payout
                }
            };
        },

            for (let type of types) {

                    results.push({
                        vehicle: result.vehicle,
                        price: result.pricing.roundedPrice,
                        originalPrice: result.pricing.finalPrice,
                        discount: result.pricing.discount,

                        savings: result.pricing.savings,

                    });
                }
            }

            results.sort((a, b) => a.price - b.price);

            return {
                success: true,
                route: {
                    distance: params.distanceKM,
                    time: params.timeHour,
                    traffic: params.trafficStatus,
                    waitMinutes: params.waitMinutes || 0
                },

            };

            this.drivers.push(driver);
            this.updateDriversUI();
            return driver;
        },

        getDriverName: function(id) {
            const d = this.drivers.find(d => d.id === id);
            return d ? d.name : 'Unknown';
        },

        getAvailableDrivers: function() {
            return this.drivers.filter(d => d.status === 'available');
        },

        // ============================================================
        // ASSIGN ORDER TO DRIVER
        // ============================================================
        assignOrder: function(orderId, driverId) {
            const order = this.orders.find(o => o.id === orderId);
            const driver = this.drivers.find(d => d.id === driverId);
            if (order && driver) {
                order.driverId = driverId;
                order.driverName = driver.name;
                order.status = 'assigned';
                driver.status = 'busy';
                driver.orders++;
                this.updateOrdersUI();
                this.updateDriversUI();
                return true;
            }
            return false;
        },

        // ============================================================
        // UPDATE ORDER STATUS
        // ============================================================
        updateOrderStatus: function(orderId, status) {
            const order = this.orders.find(o => o.id === orderId);
            if (order) {
                order.status = status;
                if (status === 'completed' || status === 'cancelled') {
                    const driver = this.drivers.find(d => d.id === order.driverId);
                    if (driver) driver.status = 'available';
                }
                this.updateOrdersUI();
                this.updateDriversUI();
                return true;
            }
            return false;
        },

    function bookOrder(vehicleType, price) {
        const customer = prompt('Enter your name:', 'Customer');
        if (!customer) return;

        const pickup = prompt('Enter pickup location:', 'Chickpet, Bengaluru');
        if (!pickup) return;

        const drop = prompt('Enter drop location:', 'Indiranagar, Bengaluru');
        if (!drop) return;

        const driverId = prompt('Enter driver ID (optional, leave blank for auto-assign):', '');
        const driver = driverId ? parseInt(driverId) : null;

        const order = ParcelMaadi.createOrder(customer, pickup, drop, vehicleType, price, driver);

        if (!driver) {
            const available = ParcelMaadi.getAvailableDrivers();
            if (available.length > 0) {
                const assigned = ParcelMaadi.assignOrder(order.id, available[0].id);
                if (assigned) showNotification('✅ Order Assigned', `Order #${order.id} assigned to ${available[0].name}`);
            } else {
                showNotification('⚠️ No Drivers', 'Order created but no drivers available. Please add drivers.');
            }
        }

        showNotification('✅ Order Created', `Order #${order.id} created for ${customer}`);

        // Switch to orders tab
        setTimeout(() => showTab('orders'), 1000);
    }

    function saveVehiclePrices() {
        document.querySelectorAll('.v-price-input').forEach(inp => {
            const key = inp.dataset.key;
            const val = parseFloat(inp.value);
            if (ParcelMaadi.vehicles[key]) ParcelMaadi.vehicles[key].baseFare = val;
        });
        syncAll();

    }

    function saveSlabRates() {
        document.querySelectorAll('.slab-input').forEach(inp => {
            const parts = inp.dataset.key.split('_');
            const vehicle = parts[0];
            const idx = parseInt(parts[1]);
            const val = parseFloat(inp.value);
            if (ParcelMaadi.slabs[vehicle] && ParcelMaadi.slabs[vehicle][idx]) {
                ParcelMaadi.slabs[vehicle][idx].rate = val;
            }
        });
        syncAll();
        showNotification('✅ Saved', 'Slab rates updated.');
    }

    function saveSurgeSettings() {
        document.querySelectorAll('.surge-input').forEach(inp => {
            const key = inp.dataset.key;
            const val = parseFloat(inp.value);
            if (key.startsWith('time_')) {
                const idx = parseInt(key.split('_')[1]);
                if (ParcelMaadi.timeSurge[idx]) ParcelMaadi.timeSurge[idx].multiplier = val;
            } else if (key.startsWith('traffic_')) {
                const tKey = key.split('_')[1];
                if (ParcelMaadi.trafficSurge[tKey] !== undefined) ParcelMaadi.trafficSurge[tKey] = val;
            }
        });
        syncAll();
        showNotification('✅ Saved', 'Surge settings updated.');
    }

    function saveDiscountSettings() {
        document.querySelectorAll('.discount-input').forEach(inp => {
            const key = inp.dataset.key;
            const val = parseFloat(inp.value);
            const type = inp.dataset.type;
            if (ParcelMaadi.discounts[key]) {
                if (type === 'value') ParcelMaadi.discounts[key].value = val;
                else if (type === 'max') ParcelMaadi.discounts[key].max = val;
            }
        });
        syncAll();
        showNotification('✅ Saved', 'Discount settings updated.');
    }

        ParcelMaadi.sync();
